# vtt-clean Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `vtt-clean`, a Python CLI that parses WebVTT files (including YouTube rolling-caption variants) into a canonical JSONL representation and emits ASR-ready formats (STM, plain text, HuggingFace manifest, Kaldi triplet) with configurable event taxonomy, Arabic-aware normalization, and corpus-level statistics.

**Architecture:** VTT → `RawCue[]` → (ordered pipeline of `Step` / `StreamStep`) → `Segment[]` (canonical) → N emitters. Pipeline is declarative, defined in YAML, with named steps registered via a decorator; a central runner enforces ordering constraints. Taxonomy maps delimiter-wrapped surfaces (`[موسيقى]`, `{...quran...}`, `(...)`) to typed events; Quran content is detected by tashkeel density and preserved as speech.

**Tech Stack:** Python 3.10+, `click`, `pyyaml`, `regex`, `tqdm`. No numpy, no pandas. `pytest` + `click.testing.CliRunner` for tests. `ruff` for lint+format. `pipx`/`pyproject.toml` for distribution. `hatchling` build backend.

**Spec:** `docs/superpowers/specs/2026-10-09-vtt-cli-design.md`

## Global Constraints

- Python version floor: 3.10. Code uses `from __future__ import annotations` and PEP 604 unions.
- Core deps: `click >=8.1`, `pyyaml >=6.0`, `regex >=2024.0`, `tqdm >=4.66`. No numpy, no pandas, no polars.
- Public CLI entry point name: `vtt-clean`. Package name: `vtt_clean`.
- Non-speech event tokens in STM/TXT are Babel-style English literals: `<music> <applause> <laugh> <singing> <cheer> <scream> <sigh> <breath> <cough> <throat> <noise> <unk>`. `aside` renders as empty string. No Arabic in output tokens.
- Default `--strip-diacritics` mode is `non-quran`. Quran content is **always** preserved verbatim regardless of flag unless mode is `all`.
- YouTube-dedup heuristic triggers only when a file contains inline `<HH:MM:SS.sss>` tags *or* cue settings matching `align:\S+ position:\S+`.
- Collated event spans (e.g., `[موسيقى][غناء]`) are always split into separate `Event` entries in canonical JSONL.
- All timestamps are seconds (float), not strings. All output file paths UTF-8.
- Test coverage floor: 85% (`pytest --cov-fail-under=85`).

from __future__ import annotations
from dataclasses import asdict, dataclass, field
from typing import Any

@dataclass
class RawCue:

    end: float
    text: str
    settings: str = ""

@dataclass

    w: str
    t: float

@dataclass

    type: str
    surface: str
    start: float
    end: float

@dataclass
class Segment:
    id: str
    source: str
    start: float
    end: float
    duration: float
    text: str
    text_raw: str
    events: list[Event] = field(default_factory=list)
    quran: bool = False
    lang: str = "ar"
    n_words: int = 0
    n_chars: int = 0
    wps: float = 0.0
    words: list[Word] | None = None
    flags: dict[str, Any] = field(default_factory=dict)

    def to_json_dict(self) -> dict[str, Any]:
        d = asdict(self)
        if self.words is None:
            d["words"] = None
        return d

`vtt_clean/config.py`:
```python
from __future__ import annotations
from dataclasses import dataclass, field
from importlib.resources import files
from pathlib import Path
from typing import Any
import yaml

class TaxonomyError(ValueError):
    pass

class PipelineError(ValueError):
    pass

@dataclass
class Taxonomy:
    events: dict[str, list[str]]
    delimiters: dict[str, str]
    rules: dict[str, Any]
    normalization: dict[str, Any]
    output_tokens: dict[str, str]
    default_lang: str
    _surface_to_event: dict[str, str] = field(default_factory=dict)

    def __post_init__(self) -> None:
        for ev, surfaces in self.events.items():
            for s in surfaces:
                if s in self._surface_to_event:
                    raise TaxonomyError(
                        f"duplicate surface {s!r} mapped to {self._surface_to_event[s]!r} and {ev!r}"
                    )
                self._surface_to_event[s] = ev
        for ev in self.events:
            if ev not in self.output_tokens:
                raise TaxonomyError(f"event {ev!r} has no output_tokens entry")

    def lookup(self, surface: str) -> str | None:
        return self._surface_to_event.get(surface.strip())

    def render_token(self, event_type: str) -> str:
        return self.output_tokens.get(event_type, "<unk>")

@dataclass
class PipelineSpec:
    steps: list[str]

def _load_yaml(path: Path) -> dict:
    with open(path, encoding="utf-8") as fh:
        return yaml.safe_load(fh)

def load_taxonomy(path: Path | None) -> Taxonomy:
    if path is None:
        data = yaml.safe_load(files("vtt_clean").joinpath("default_taxonomy.yaml").read_text(encoding="utf-8"))
    else:
        data = _load_yaml(path)
    try:
        return Taxonomy(
            events=data["events"],
            delimiters=data["delimiters"],
            rules=data["rules"],
            normalization=data["normalization"],
            output_tokens=data["output_tokens"],
            default_lang=data.get("default_lang", "ar"),
        )
    except KeyError as e:
        raise TaxonomyError(f"missing required taxonomy key: {e}") from e

def load_pipeline(path: Path | None) -> PipelineSpec:
    if path is None:
        data = yaml.safe_load(files("vtt_clean").joinpath("default_pipeline.yaml").read_text(encoding="utf-8"))
    else:
        data = _load_yaml(path)
    steps = data.get("steps")
    if not steps or not isinstance(steps, list):
        raise PipelineError("pipeline YAML must have a non-empty 'steps' list")
    return PipelineSpec(steps=steps)
```

`vtt_clean/parse/vtt.py`:
```python
from __future__ import annotations
import re
from collections.abc import Iterator
from pathlib import Path
from vtt_clean.model import RawCue

_TS_LINE = re.compile(
    r"^\s*(\d\d:)?\d\d:\d\d\.\d{3}\s*-->\s*(\d\d:)?\d\d:\d\d\.\d{3}(\s+.*)?$"
)

def _ts_to_sec(ts: str) -> float:
    parts = ts.strip().split(":")
    if len(parts) == 3:
        h, m, s = parts
        return int(h) * 3600 + int(m) * 60 + float(s)
    m, s = parts
    return int(m) * 60 + float(s)

def parse_vtt(path: Path) -> Iterator[RawCue]:
    with open(path, encoding="utf-8") as fh:
        lines = fh.read().splitlines()
    i = 0
    in_header = True
    while i < len(lines):
        line = lines[i]
        if in_header:
            if "-->" in line and _TS_LINE.match(line):
                in_header = False
            else:
                i += 1
                continue
        if "-->" in line and _TS_LINE.match(line):
            left, right = line.split("-->", 1)
            right = right.strip()
            rparts = right.split(None, 1)
            end_ts = rparts[0]
            settings = rparts[1] if len(rparts) > 1 else ""
            start = _ts_to_sec(left)
            end = _ts_to_sec(end_ts)
            i += 1
            text_lines: list[str] = []
            while i < len(lines) and lines[i].strip() != "" and "-->" not in lines[i]:
                text_lines.append(lines[i])
                i += 1
            yield RawCue(start=start, end=end, text="\n".join(text_lines), settings=settings)
        else:
            i += 1
```

from __future__ import annotations
from collections.abc import Iterable, Iterator
from dataclasses import dataclass, field
from typing import Any, Protocol, runtime_checkable
from vtt_clean.config import Taxonomy
from vtt_clean.model import RawCue, Segment

@dataclass
class PipelineContext:
    taxonomy: Taxonomy
    source: str
    lang: str
    emit_word_timings: bool
    strip_diacritics_mode: str  # "none" | "non-quran" | "all"
    keep_punctuation: bool
    keep_symbols: bool
    extras: dict[str, Any] = field(default_factory=dict)

@runtime_checkable
class Step(Protocol):
    name: str
    def apply(self, seg: Segment, ctx: PipelineContext) -> Segment: ...

@runtime_checkable
class StreamStep(Protocol):
    name: str
    def apply_stream(self, cues: Iterable[RawCue], ctx: PipelineContext) -> Iterator[RawCue]: ...

_STEP_REGISTRY: dict[str, type] = {}
_STREAM_REGISTRY: dict[str, type] = {}

def register_step(name: str):
    def deco(cls):
        _STEP_REGISTRY[name] = cls
        return cls
    return deco

def register_stream_step(name: str):
    def deco(cls):
        _STREAM_REGISTRY[name] = cls
        return cls
    return deco

def get_step(name: str) -> type | None:
    return _STEP_REGISTRY.get(name)

def get_stream_step(name: str) -> type | None:
    return _STREAM_REGISTRY.get(name)

def all_step_names() -> set[str]:
    return set(_STEP_REGISTRY) | set(_STREAM_REGISTRY)

from __future__ import annotations
from collections.abc import Iterable
from dataclasses import dataclass
from vtt_clean.config import PipelineError, PipelineSpec
from vtt_clean.model import RawCue, Segment
from vtt_clean.pipeline.base import (
    PipelineContext, all_step_names, get_step, get_stream_step,
)

# Ordering constraints: {step_that_must_come_first: {steps_that_must_come_after}}
ORDERING: dict[str, set[str]] = {
    "extract_word_timings": {"youtube_dedup"},
    "extract_events": {"strip_emoji", "strip_diacritics", "strip_punctuation", "normalize_whitespace"},
    "quran_detect": {"strip_diacritics"},
}

@dataclass
class Runner:
    stream_steps: list
    seg_steps: list

    def run(self, cues: Iterable[RawCue], ctx: PipelineContext) -> list[Segment]:
        stream = iter(cues)
        for s in self.stream_steps:
            stream = s.apply_stream(stream, ctx)
        segments: list[Segment] = []
        for idx, cue in enumerate(stream):
            seg = Segment(
                id=f"{_stem(ctx.source)}-{idx:04d}",
                source=ctx.source,
                start=cue.start, end=cue.end, duration=max(0.0, cue.end - cue.start),
                text=cue.text, text_raw=cue.text,
                lang=ctx.lang,
                flags={"youtube_dedup_applied": False, "had_karaoke_timing": False,
                       "had_cue_settings": bool(cue.settings), "unknown_events": []},
            )
            for s in self.seg_steps:
                seg = s.apply(seg, ctx)
            segments.append(seg)
        return segments

def _stem(source: str) -> str:
    base = source.rsplit("/", 1)[-1]
    for suffix in (".ar.vtt", ".vtt"):
        if base.endswith(suffix):
            return base[: -len(suffix)]
    return base

def build_runner(spec: PipelineSpec) -> Runner:
    known = all_step_names()
    for name in spec.steps:
        if name not in known:
            raise PipelineError(f"unknown step: {name!r}")
    # Enforce ordering.
    positions = {n: i for i, n in enumerate(spec.steps)}
    for before, afters in ORDERING.items():
        if before not in positions:
            continue
        for after in afters:
            if after in positions and positions[after] <= positions[before]:
                raise PipelineError(
                    f"order violation: {before!r} must come before {after!r}"
                )
    stream_steps = []
    seg_steps = []
    for name in spec.steps:
        if (cls := get_stream_step(name)) is not None:
            stream_steps.append(cls())
        else:
            cls = get_step(name)
            seg_steps.append(cls())
    return Runner(stream_steps=stream_steps, seg_steps=seg_steps)
```

from __future__ import annotations
import re
from collections.abc import Iterable, Iterator
from vtt_clean.model import RawCue
from vtt_clean.pipeline.base import PipelineContext, register_stream_step

_INLINE_TS = re.compile(r"<(\d\d):(\d\d):(\d\d\.\d{3})>")
_C_OPEN = re.compile(r"<c[^>]*>")
_C_CLOSE = re.compile(r"</c>")

def _ts_to_sec(h: str, m: str, s: str) -> float:
    return int(h) * 3600 + int(m) * 60 + float(s)

@register_stream_step("extract_word_timings")
class ExtractWordTimings:
    name = "extract_word_timings"

    def apply_stream(self, cues: Iterable[RawCue], ctx: PipelineContext) -> Iterator[RawCue]:
        for idx, cue in enumerate(cues):
            text = cue.text
            had_karaoke = bool(_INLINE_TS.search(text) or _C_OPEN.search(text))
            if had_karaoke:
                ctx.extras["had_karaoke_timing"] = True
                if ctx.emit_word_timings:
                    words = _extract_words(text)
                    ctx.extras[f"words_{idx}"] = words
            # Strip tags
            text = _INLINE_TS.sub("", text)
            text = _C_OPEN.sub("", text)
            text = _C_CLOSE.sub("", text)
            cue.text = text
            yield cue

def _extract_words(text: str) -> list[dict]:
    # Pattern: <ts><c> word</c>
    pattern = re.compile(r"<(\d\d):(\d\d):(\d\d\.\d{3})><c[^>]*>\s*([^<]+?)\s*</c>")
    return [
        {"w": m.group(4), "t": _ts_to_sec(m.group(1), m.group(2), m.group(3))}
        for m in pattern.finditer(text)
    ]
```

from __future__ import annotations
import math
from collections.abc import Iterable, Iterator
from vtt_clean.model import RawCue
from vtt_clean.pipeline.base import PipelineContext, register_stream_step

@register_stream_step("youtube_dedup")
class YouTubeDedup:
    name = "youtube_dedup"

    def apply_stream(self, cues: Iterable[RawCue], ctx: PipelineContext) -> Iterator[RawCue]:
        buf = list(cues)
        triggered = ctx.extras.get("had_karaoke_timing") or ctx.extras.get("had_cue_settings")
        if not triggered:
            yield from buf
            return
        kept: list[RawCue] = []
        removed_chars = 0
        for cue in buf:
            if _is_spacer(cue, kept[-1] if kept else None):
                removed_chars += len(cue.text)
                continue
            if kept:
                overlap = _overlap_len(kept[-1].text, cue.text)
                if overlap > 0:
                    removed_chars += overlap
                    cue.text = cue.text[overlap:].lstrip()
                    if cue.text == "":
                        continue
            cue.text = cue.text  # unchanged if no overlap
            kept.append(cue)
        ctx.extras["youtube_dedup_applied"] = True
        ctx.extras["dedup_removed_chars"] = ctx.extras.get("dedup_removed_chars", 0) + removed_chars
        yield from kept

def _is_spacer(cue: RawCue, prev: RawCue | None) -> bool:
    dur = cue.end - cue.start
    if dur <= 0.020 and cue.text.strip() == "":
        return True
    if prev is not None and cue.text.strip() == prev.text.strip() and dur <= 0.050:
        return True
    return False

def _overlap_len(prev: str, curr: str) -> int:
    """Longest common prefix of curr with any suffix of prev, returned as a
    character count on curr. Must be >= ceil(min(len)/2) to count."""
    prev_s = prev.strip()
    curr_s = curr.strip()
    if not prev_s or not curr_s:
        return 0
    max_candidate = min(len(prev_s), len(curr_s))
    threshold = math.ceil(min(len(prev_s), len(curr_s)) / 2)
    # Try overlap sizes from longest to shortest.
    for n in range(max_candidate, threshold - 1, -1):
        if prev_s.endswith(curr_s[:n]):
            # Map back to position in original (unstripped) curr.
            lead = len(curr) - len(curr.lstrip())
            return lead + n
    return 0
```

from __future__ import annotations
import regex as re
from vtt_clean.model import Event, Segment
from vtt_clean.pipeline.base import PipelineContext, register_step
from vtt_clean.util.arabic import count_tashkeel

# Spans with surrounding delimiters captured; whitespace handling preserved.
_SPAN = re.compile(r"(\[[^\[\]]*\]|\{[^\{\}]*\}|\([^\(\)]*\))")

@register_step("extract_events")
class ExtractEvents:
    name = "extract_events"

    def apply(self, seg: Segment, ctx: PipelineContext) -> Segment:
        text = seg.text_raw
        matches = list(_SPAN.finditer(text))
        if not matches:
            seg.text = text
            return seg
        min_tashkeel = ctx.taxonomy.rules.get("quran_detection", {}).get("min_diacritics", 2)
        events: list[Event] = []
        # Determine positions, considering adjacency for collated split.
        text_wo_spaces = text.strip()
        standalone_span = len(matches) == 1 and matches[0].group(0).strip() == text_wo_spaces
        # Build replacement intervals (start, end, replacement_text, kind)
        replacements: list[tuple[int, int, str]] = []
        for i, m in enumerate(matches):
            surface = m.group(0)
            inner = surface[1:-1].strip()
            delim = surface[0]
            keep_in_text = False
            if delim == "[":
                ev_type = ctx.taxonomy.lookup(inner) or "unk"
            elif delim == "{":
                if count_tashkeel(inner) >= min_tashkeel:
                    # Quran: keep in text, emit no event here (quran_detect handles flag).
                    continue
                ev_type = ctx.taxonomy.lookup(inner) or "unk"
            else:  # (
                ev_type = "aside"
            if ev_type == "unk":
                seg.flags.setdefault("unknown_events", []).append(surface)
            position = _position(standalone_span, i, len(matches), m, text)
            events.append(Event(type=ev_type, surface=surface,
                                start=seg.start, end=seg.end, position=position))
            replacements.append((m.start(), m.end(), ""))
            _ = keep_in_text  # silence linter
        # Apply replacements right-to-left.
        new_text = text
        for s, e, rep in sorted(replacements, key=lambda t: -t[0]):
            new_text = new_text[:s] + rep + new_text[e:]
        seg.text = new_text
        seg.events = events
        return seg

def _position(standalone: bool, i: int, total: int, m, text: str) -> str:
    if standalone:
        return "standalone"
    if i == 0 and text[: m.start()].strip() == "":
        return "pre"
    if i == total - 1 and text[m.end():].strip() == "":
        return "post"
    return "mid"
```

from __future__ import annotations
import regex as re
from vtt_clean.model import Segment
from vtt_clean.pipeline.base import PipelineContext, register_step
from vtt_clean.util.arabic import count_tashkeel

_BRACE = re.compile(r"\{([^\{\}]*)\}")

@register_step("quran_detect")
class QuranDetect:
    name = "quran_detect"

    def apply(self, seg: Segment, ctx: PipelineContext) -> Segment:
        min_t = ctx.taxonomy.rules.get("quran_detection", {}).get("min_diacritics", 2)
        matched = False

        def _sub(m):
            nonlocal matched
            inner = m.group(1)
            if count_tashkeel(inner) >= min_t:
                matched = True
                return inner
            return m.group(0)

        seg.text = _BRACE.sub(_sub, seg.text)
        if matched:
            seg.quran = True
        return seg
```

from typing import TextIO
from vtt_clean.config import Taxonomy, load_taxonomy
from vtt_clean.model import Segment

def render_text_with_events(seg: Segment, taxonomy: Taxonomy) -> str:
    pre: list[str] = []
    post: list[str] = []
    for e in seg.events:
        tok = taxonomy.render_token(e.type)
        if not tok:
            continue
        if e.position == "pre":
            pre.append(tok)
        elif e.position == "standalone":
            pre.append(tok)  # becomes the whole line
        else:  # mid, post → appended
            post.append(tok)
    parts = pre + ([seg.text] if seg.text else []) + post
    return " ".join(p for p in parts if p)

def emit_txt(segments: Iterable[Segment], out: TextIO, taxonomy: Taxonomy | None = None) -> None:
    tax = taxonomy or load_taxonomy(None)
    for seg in segments:
        out.write(render_text_with_events(seg, tax))
        out.write("\n")

from __future__ import annotations
from collections.abc import Iterable
from typing import TextIO
from vtt_clean.config import Taxonomy, load_taxonomy
from vtt_clean.emit.txt import render_text_with_events
from vtt_clean.model import Segment

def _stem(source: str) -> str:
    base = source.rsplit("/", 1)[-1]
    for suffix in (".ar.vtt", ".vtt"):
        if base.endswith(suffix):
            return base[: -len(suffix)]
    return base

def emit_stm(
    segments: Iterable[Segment],
    out: TextIO,
    taxonomy: Taxonomy | None = None,
    spk: str = "unknown",
    channel: str = "1",
) -> None:
    tax = taxonomy or load_taxonomy(None)
    for seg in segments:
        text = render_text_with_events(seg, tax)
        if not text:
            continue
        out.write(f"{_stem(seg.source)} {channel} {spk} {seg.start:.3f} {seg.end:.3f} {text}\n")

        "\n".join(f"{r} # placeholder — populate with path to audio" for r in sorted(recos)) + "\n",

from __future__ import annotations
import statistics as st
from collections import Counter
from vtt_clean.model import Segment

def _quantile(xs: list[float], q: float) -> float:
    if not xs:
        return 0.0
    xs = sorted(xs)
    k = (len(xs) - 1) * q
    f = int(k); c = min(f + 1, len(xs) - 1)
    return xs[f] + (xs[c] - xs[f]) * (k - f)

def compute_file_metrics(segments: list[Segment], removed_chars: int = 0) -> dict:
    wps = [s.wps for s in segments if s.wps > 0]
    wps_per_seg = [s.n_words for s in segments]
    durs = [s.duration for s in segments]
    return {
        "file": segments[0].source if segments else "",
        "n_segments": len(segments),
        "n_words": sum(s.n_words for s in segments),
        "total_speech_sec": sum(durs),
        "total_event_only_sec": sum(s.duration for s in segments if not s.text.strip() and s.events),
        "wps_mean": (st.mean(wps) if wps else 0.0),
        "wps_median": (st.median(wps) if wps else 0.0),
        "words_per_seg_mean": (st.mean(wps_per_seg) if wps_per_seg else 0.0),
        "words_per_seg_p95": _quantile([float(x) for x in wps_per_seg], 0.95),
        "words_per_seg_max": (max(wps_per_seg) if wps_per_seg else 0),
        "duration_mean": (st.mean(durs) if durs else 0.0),
        "duration_p95": _quantile(durs, 0.95),
        "n_events_total": sum(len(s.events) for s in segments),
        "n_quran_segments": sum(1 for s in segments if s.quran),
        "n_unknown_events": sum(len(s.flags.get("unknown_events", [])) for s in segments),
        "youtube_dedup_removed_dup_chars": removed_chars,
    }

def aggregate_corpus(
    file_metrics: list[dict],
    per_file_segments: dict[str, list[Segment]],
    top_n_events: int = 50,
    top_n_unknown: int = 100,
    vocab: set[str] | None = None,
) -> dict:
    agg = {
        "n_files": len(file_metrics),
        "n_segments": sum(m["n_segments"] for m in file_metrics),
        "n_words": sum(m["n_words"] for m in file_metrics),
        "total_speech_sec": sum(m["total_speech_sec"] for m in file_metrics),
        "n_events_total": sum(m["n_events_total"] for m in file_metrics),
        "n_quran_segments": sum(m["n_quran_segments"] for m in file_metrics),
        "youtube_dedup_removed_dup_chars": sum(m["youtube_dedup_removed_dup_chars"] for m in file_metrics),
    }
    ev_counter: Counter = Counter()
    unk_counter: Counter = Counter()
    unk_examples: dict[str, set] = {}
    oov = 0; total_tok = 0
    for fname, segs in per_file_segments.items():
        for s in segs:
            for e in s.events:
                ev_counter[e.type] += 1
                if e.type == "unk":
                    unk_counter[e.surface] += 1
                    unk_examples.setdefault(e.surface, set()).add(fname)
            if vocab is not None:
                for tok in s.text.split():
                    total_tok += 1
                    if tok not in vocab:
                        oov += 1
    agg["event_type_counts"] = dict(ev_counter.most_common(top_n_events))
    agg["unknown_surface_counts"] = [
        {"surface": s, "count": c, "example_files": sorted(unk_examples.get(s, set()))[:3]}
        for s, c in unk_counter.most_common(top_n_unknown)
    ]
    if vocab is not None:
        agg["oov_tokens"] = oov
        agg["total_tokens"] = total_tok
        agg["oov_rate"] = (oov / total_tok) if total_tok else 0.0
    return agg
```

def write_stats_tsv(rows: list[dict], path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    if not rows:
        path.write_text("", encoding="utf-8")
        return
    keys = sorted(rows[0].keys())
    out = ["\t".join(keys)]
    for r in rows:
        out.append("\t".join(str(r.get(k, "")) for k in keys))
    path.write_text("\n".join(out) + "\n", encoding="utf-8")

def write_unknown_events_tsv(aggregated: dict, path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    rows = ["surface\tcount\texample_files"]

        rows.append(f"{r['surface']}\t{r['count']}\t{','.join(r['example_files'])}")

from __future__ import annotations
import importlib
from dataclasses import dataclass
from pathlib import Path
from vtt_clean.config import PipelineSpec, load_pipeline, load_taxonomy
from vtt_clean.emit.hf import emit_hf
from vtt_clean.emit.jsonl import emit_jsonl
from vtt_clean.emit.kaldi import emit_kaldi
from vtt_clean.emit.stm import _stem, emit_stm
from vtt_clean.emit.txt import emit_txt
from vtt_clean.parse.vtt import parse_vtt
from vtt_clean.pipeline.base import PipelineContext
from vtt_clean.pipeline.runner import build_runner
from vtt_clean.stats.metrics import compute_file_metrics

# Eagerly import step modules so @register_step runs
for _m in ("strip_cue_settings", "extract_word_timings", "youtube_dedup",
           "extract_events", "quran_detect", "strip_emoji",
           "strip_diacritics", "strip_punctuation", "normalize_whitespace",
           "compute_metrics"):
    importlib.import_module(f"vtt_clean.pipeline.steps.{_m}")

ALL_FORMATS = {"jsonl", "stm", "txt", "hf", "kaldi"}

@dataclass
class ProcessConfig:
    taxonomy_path: Path | None = None
    pipeline_path: Path | None = None
    emit_word_timings: bool = False
    no_youtube_dedup: bool = False
    strip_diacritics: str = "non-quran"
    keep_punctuation: bool = False
    keep_symbols: bool = False
    lang: str = "ar"

def process_one(path: Path, out_dir: Path, formats: set[str], cfg: ProcessConfig):
    taxonomy = load_taxonomy(cfg.taxonomy_path)
    spec = load_pipeline(cfg.pipeline_path)
    steps = [s for s in spec.steps if not (cfg.no_youtube_dedup and s == "youtube_dedup")]
    runner = build_runner(PipelineSpec(steps=steps))
    ctx = PipelineContext(
        taxonomy=taxonomy, source=path.name, lang=cfg.lang,
        emit_word_timings=cfg.emit_word_timings,
        strip_diacritics_mode=cfg.strip_diacritics,
        keep_punctuation=cfg.keep_punctuation,
        keep_symbols=cfg.keep_symbols,
        extras={},
    )
    cues = parse_vtt(path)
    segments = runner.run(cues, ctx)
    # Attach word timings when flagged.
    if cfg.emit_word_timings:
        for idx, seg in enumerate(segments):
            w = ctx.extras.get(f"words_{idx}")
            if w:
                seg.words = w
    stem = _stem(str(path))
    # Emitters per format.
    if "jsonl" in formats:
        d = out_dir / "jsonl"; d.mkdir(parents=True, exist_ok=True)
        with open(d / f"{stem}.jsonl", "w", encoding="utf-8") as fh:
            emit_jsonl(segments, fh)
    if "stm" in formats:
        d = out_dir / "stm"; d.mkdir(parents=True, exist_ok=True)
        with open(d / f"{stem}.stm", "w", encoding="utf-8") as fh:
            emit_stm(segments, fh, taxonomy)
    if "txt" in formats:
        d = out_dir / "txt"; d.mkdir(parents=True, exist_ok=True)
        with open(d / f"{stem}.txt", "w", encoding="utf-8") as fh:
            emit_txt(segments, fh, taxonomy)
    if "hf" in formats:
        d = out_dir / "hf"; d.mkdir(parents=True, exist_ok=True)
        with open(d / "manifest.jsonl", "a", encoding="utf-8") as fh:
            emit_hf(segments, fh)
    if "kaldi" in formats:
        emit_kaldi(segments, out_dir / "kaldi", taxonomy)
    removed = ctx.extras.get("dedup_removed_chars", 0)
    metrics = compute_file_metrics(segments, removed_chars=removed)
    return metrics, segments
```

from __future__ import annotations
from pathlib import Path
import click
from vtt_clean.driver import ALL_FORMATS, ProcessConfig, process_one
from vtt_clean.stats.metrics import aggregate_corpus
from vtt_clean.stats.reports import (
    write_stats_json, write_stats_tsv, write_unknown_events_tsv,
)

def _resolve_inputs(inputs: tuple[str, ...], glob: str) -> list[Path]:
    out: list[Path] = []
    for inp in inputs:
        p = Path(inp)
        if p.is_dir():
            out.extend(sorted(p.rglob(glob)))
        elif p.is_file():
            out.append(p)
        else:
            out.extend(sorted(Path().glob(inp)))
    return out

@click.group()
@click.version_option()
def main() -> None:
    """vtt-clean: VTT cleaning and text extraction for ASR corpora."""

@main.command()
@click.argument("inputs", nargs=-1, required=True)
@click.option("-o", "--out-dir", type=click.Path(path_type=Path), default=Path("out"))
@click.option("--glob", default="*.vtt", show_default=True)
@click.option("--format", "formats", multiple=True,
              type=click.Choice(["jsonl", "stm", "txt", "hf", "kaldi", "all"]),
              default=["all"], show_default=True)
@click.option("--config", type=click.Path(exists=True, path_type=Path), default=None)
@click.option("--taxonomy", type=click.Path(exists=True, path_type=Path), default=None)
@click.option("--emit-word-timings", is_flag=True)
@click.option("--no-youtube-dedup", is_flag=True)
@click.option("--strip-diacritics",
              type=click.Choice(["none", "non-quran", "all"]), default="non-quran",
              show_default=True)
@click.option("--keep-punctuation", is_flag=True)
@click.option("--keep-symbols", is_flag=True)
@click.option("--no-stats", is_flag=True)
@click.option("--vocab", type=click.Path(exists=True, path_type=Path), default=None)
@click.option("--lang", default="ar", show_default=True)
def process(inputs, out_dir, glob, formats, config, taxonomy, emit_word_timings,
            no_youtube_dedup, strip_diacritics, keep_punctuation, keep_symbols,
            no_stats, vocab, lang):
    """Clean VTTs and emit one or more formats."""
    files = _resolve_inputs(inputs, glob)
    if not files:
        raise click.ClickException("no VTT files found")
    fmts: set[str] = set()
    for f in formats:
        if f == "all":
            fmts.update(ALL_FORMATS)
        else:
            fmts.add(f)
    cfg = ProcessConfig(
        taxonomy_path=taxonomy, pipeline_path=config,
        emit_word_timings=emit_word_timings, no_youtube_dedup=no_youtube_dedup,
        strip_diacritics=strip_diacritics, keep_punctuation=keep_punctuation,
        keep_symbols=keep_symbols, lang=lang,
    )
    out_dir.mkdir(parents=True, exist_ok=True)
    # Fresh HF manifest if emitted.
    if "hf" in fmts:
        (out_dir / "hf").mkdir(parents=True, exist_ok=True)
        (out_dir / "hf" / "manifest.jsonl").write_text("", encoding="utf-8")
    file_metrics = []
    per_file_segments: dict[str, list] = {}
    for p in files:
        m, segs = process_one(p, out_dir, fmts, cfg)
        file_metrics.append(m)
        per_file_segments[p.name] = segs
    if not no_stats:
        vocab_set: set[str] | None = None
        if vocab is not None:
            vocab_set = set(Path(vocab).read_text(encoding="utf-8").split())
        agg = aggregate_corpus(file_metrics, per_file_segments, vocab=vocab_set)
        write_stats_json(agg, out_dir / "stats.json")
        write_stats_tsv(file_metrics, out_dir / "stats.tsv")
        write_unknown_events_tsv(agg, out_dir / "unknown-events.tsv")
    click.echo(f"processed {len(files)} file(s) → {out_dir}")

import json
import sys
import yaml
from vtt_clean.config import PipelineSpec, TaxonomyError, load_pipeline, load_taxonomy

@main.command()
@click.argument("file", type=click.Path(exists=True, path_type=Path))
@click.option("--stage", default=None, help="Halt after named step")
def inspect(file: Path, stage: str | None):
    """Dump canonical JSONL for one file to stdout."""
    taxonomy = load_taxonomy(None)
    spec = load_pipeline(None)
    steps = spec.steps
    if stage is not None:
        if stage not in steps:
            raise click.ClickException(f"unknown stage: {stage!r}")
        steps = steps[: steps.index(stage) + 1]
    runner = build_runner(PipelineSpec(steps=steps))
    ctx = PipelineContext(
        taxonomy=taxonomy, source=file.name, lang="ar",
        emit_word_timings=False, strip_diacritics_mode="non-quran",
        keep_punctuation=False, keep_symbols=False, extras={},
    )
    for seg in runner.run(parse_vtt(file), ctx):
        click.echo(json.dumps(seg.to_json_dict(), ensure_ascii=False))


@main.group()
def taxonomy() -> None:
    """Taxonomy management."""

@taxonomy.command("show")
@click.option("--taxonomy", "taxonomy_path", type=click.Path(exists=True, path_type=Path), default=None)
def tax_show(taxonomy_path: Path | None):
    tax = load_taxonomy(taxonomy_path)
    for ev, surfaces in sorted(tax.events.items()):
        tok = tax.render_token(ev)
        click.echo(f"{ev:14}  token={tok!r}  surfaces={surfaces}")

@taxonomy.command("validate")
@click.argument("path", type=click.Path(exists=True, path_type=Path))
def tax_validate(path: Path):
    try:
        load_taxonomy(path)
    except TaxonomyError as e:
        raise click.ClickException(str(e))
    click.echo("ok")

@taxonomy.command("extend")
@click.argument("path", type=click.Path(exists=True, path_type=Path))
@click.option("--from", "src", type=click.Path(exists=True, path_type=Path), required=True)
@click.option("--event", required=True)
def tax_extend(path: Path, src: Path, event: str):
    data = yaml.safe_load(path.read_text(encoding="utf-8"))
    if event not in data["events"]:
        raise click.ClickException(f"event {event!r} not declared in taxonomy")
    surfaces = []
    for line in src.read_text(encoding="utf-8").splitlines()[1:]:  # skip header
        if not line.strip():
            continue
        surface = line.split("\t", 1)[0].strip("[](){} ")
        if surface and surface not in data["events"][event]:
            surfaces.append(surface)
    data["events"][event].extend(surfaces)
    path.write_text(yaml.safe_dump(data, allow_unicode=True, sort_keys=False), encoding="utf-8")
    click.echo(f"added {len(surfaces)} surface(s) to event {event!r}")


@main.command("stats")
@click.argument("inputs", nargs=-1, required=True)
@click.option("--glob", default="*.vtt", show_default=True)
@click.option("--out", "out_path", type=click.Path(path_type=Path), default=Path("stats.json"))
def stats_cmd(inputs, glob, out_path: Path):
    files = _resolve_inputs(inputs, glob)
    if not files:
        raise click.ClickException("no VTT files found")
    cfg = ProcessConfig()
    file_metrics = []
    per_file_segments: dict[str, list] = {}
    with click.progressbar(files, label="stats") as bar:
        for p in bar:
            m, segs = process_one(p, Path("/tmp/_vtt_clean_discard"), set(), cfg)
            file_metrics.append(m)
            per_file_segments[p.name] = segs
    from vtt_clean.stats.metrics import aggregate_corpus
    from vtt_clean.stats.reports import write_stats_json
    agg = aggregate_corpus(file_metrics, per_file_segments)
    write_stats_json(agg, out_path)
```

import json
from pathlib import Path
from vtt_clean.driver import ProcessConfig, process_one

FIX = Path(__file__).parent / "fixtures"

def _collect():
    return sorted(FIX.rglob("*.vtt"))

def _run(path: Path):
    _, segs = process_one(path, Path("/tmp/_discard"), set(), ProcessConfig())
    return [s.to_json_dict() for s in segs]

def _normalize(obj):
    return json.loads(json.dumps(obj, ensure_ascii=False, sort_keys=True))

def test_golden_fixtures(update_goldens):
    for vtt in _collect():
        expected_path = vtt.with_suffix(".expected.jsonl")
        actual = _run(vtt)
        if update_goldens:
            with open(expected_path, "w", encoding="utf-8") as fh:
                for row in actual:
                    fh.write(json.dumps(row, ensure_ascii=False) + "\n")
            continue
        assert expected_path.exists(), f"missing golden for {vtt}"
        expected = [json.loads(l) for l in expected_path.read_text(encoding="utf-8").splitlines() if l.strip()]
        assert _normalize(actual) == _normalize(expected), f"golden mismatch: {vtt}"
